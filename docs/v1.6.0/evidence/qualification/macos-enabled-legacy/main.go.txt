const logicalLabel = "com.opensocrates.auto-update"

type mapping struct{ Root, Home, Label string }
type fixture struct {
	Root               string `json:"root"`
	Installed, Enabled bool
}

func must(e error) {
	if e != nil {
		panic(e)
	}
}
func hash(b []byte) string   { x := sha256.Sum256(b); return "sha256:" + hex.EncodeToString(x[:]) }
func read(p string) []byte   { b, e := os.ReadFile(p); must(e); return b }
func digest(p string) string { return hash(read(p)) }
func write(p string, b []byte, mode os.FileMode) {
	must(platform.Mkdir(filepath.Dir(p)))
	must(platform.AtomicWrite(p, b, mode))
}
func emit(v any) { b, e := contracts.CanonicalJSON(v); must(e); _, e = os.Stdout.Write(b); must(e) }
func command(ctx context.Context, exe string, args []string, env []string) ([]byte, error) {
	ctx, cancel := context.WithTimeout(ctx, 30*time.Second)
	defer cancel()
	c := exec.CommandContext(ctx, exe, args...)
	c.Env = env
	var out, stderr bytes.Buffer
	c.Stdout = &out
	c.Stderr = &stderr
	e := c.Run()
	if out.Len() > 1<<20 || stderr.Len() > 1<<20 {
		return nil, errors.New("output_bound")
	}
	if e != nil {
		return append(out.Bytes(), stderr.Bytes()...), e
	}
	return out.Bytes(), nil
}
func environment(values map[string]string) []string {
	env := []string{}
	for _, s := range os.Environ() {
		k, _, _ := strings.Cut(s, "=")
		if _, ok := values[k]; !ok {
			env = append(env, s)
		}
	}
	for k, v := range values {
		env = append(env, k+"="+v)
	}
	return env
}
func hostFixture(args []string) {
	root := os.Getenv("OPENSOCRATES_QUALIFICATION_ROOT")
	if !strings.HasPrefix(filepath.Base(root), "opensocrates-enabled-") {
		os.Exit(2)
	}
	p := filepath.Join(root, "fixture-codex.json")
	var s fixture
	if b, e := os.ReadFile(p); e == nil {
		must(json.Unmarshal(b, &s))
	}
	save := func() { b, e := json.Marshal(s); must(e); write(p, b, 0600) }
	switch {
	case len(args) >= 2 && args[0] == "login" && args[1] == "status":
		fmt.Println("Logged in using disposable qualification fixture")
		return
	case len(args) > 0 && args[0] == "--version":
		fmt.Println("codex-cli 0.130.0")
		return
	case len(args) >= 3 && args[0] == "plugin" && args[1] == "marketplace":
		switch args[2] {
		case "list":
			rows := []any{}
			if s.Root != "" {
				rows = append(rows, map[string]any{"name": "opensocrates", "root": s.Root})
			}
			emit(map[string]any{"marketplaces": rows})
			return
		case "add":
			if len(args) < 4 {
				os.Exit(2)
			}
			s.Root = args[3]
			save()
			emit(map[string]any{"name": "opensocrates", "root": s.Root})
			return
		case "remove":
			s.Root = ""
			save()
			emit(map[string]any{"removed": true})
			return
		}
	case len(args) >= 2 && args[0] == "plugin":
		switch args[1] {
		case "list":
			rows := []any{}
			if s.Installed {
				rows = append(rows, map[string]any{"pluginId": "opensocrates@opensocrates", "version": "1.5.0", "enabled": s.Enabled})
			}
			emit(map[string]any{"installed": rows, "available": []any{}})
			return
		case "add":
			s.Installed = true
			s.Enabled = true
			save()
			emit(map[string]any{"pluginId": "opensocrates@opensocrates", "version": "1.5.0", "enabled": true})
			return
		case "remove":
			s.Installed = false
			save()
			emit(map[string]any{"removed": true})
			return
		}
	}
	os.Exit(2)
}
func (a mapping) domain() string { return "gui/" + strconv.Itoa(os.Getuid()) }
func (a mapping) target() string { return a.domain() + "/" + a.Label }
func (a mapping) canonical() string {
	return filepath.Join(a.Home, "Library", "LaunchAgents", logicalLabel+".plist")
}
func (a mapping) mapped() string { return filepath.Join(a.Root, "actual-owned-unique.plist") }
func (a mapping) print(ctx context.Context) ([]byte, error) {
	return command(ctx, "/bin/launchctl", []string{"print", a.target()}, os.Environ())
}
func (a mapping) run(ctx context.Context, exe string, args ...string) ([]byte, error) {
	if exe != "/bin/launchctl" && exe != "launchctl" {
		return nil, errors.New("unowned_scheduler_tool")
	}
	switch {
	case len(args) == 2 && args[0] == "print" && args[1] == a.domain()+"/"+logicalLabel:
		return a.print(ctx)
	case len(args) == 2 && args[0] == "print" && args[1] == a.domain():
		b, e := a.print(ctx)
		if e == nil {
			return []byte("qualification-owned " + logicalLabel + "\n"), nil
		}
		if !bytes.Contains(b, []byte("Could not find service")) || !bytes.Contains(b, []byte(a.Label)) {
			return nil, errors.New("unknown_owned_job_state")
		}
		return []byte("qualification-owned absent\n"), nil
	case len(args) == 2 && args[0] == "bootout" && args[1] == a.domain()+"/"+logicalLabel:
		return command(ctx, "/bin/launchctl", []string{"bootout", a.target()}, os.Environ())
	case len(args) == 3 && args[0] == "bootstrap" && args[1] == a.domain() && args[2] == a.canonical():
		b, e := platform.ReadOwnedFile(a.canonical(), 1<<20)
		if e != nil {
			return nil, e
		}
		label := regexp.MustCompile(`(<key>\s*Label\s*</key>\s*<string>)com\.opensocrates\.auto-update(</string>)`)
		start := regexp.MustCompile(`(<key>\s*RunAtLoad\s*</key>\s*)<true/>`)
		if len(label.FindAll(b, -1)) != 1 || len(start.FindAll(b, -1)) != 1 {
			return nil, errors.New("unknown_owned_plist")
		}
		b = label.ReplaceAll(b, []byte("${1}"+a.Label+"${2}"))
		b = start.ReplaceAll(b, []byte("${1}<false/>"))
		write(a.mapped(), b, 0600)
		return command(ctx, "/bin/launchctl", []string{"bootstrap", a.domain(), a.mapped()}, os.Environ())
	}
	return nil, errors.New("unowned_scheduler_arguments")
}
func plistArgs(b []byte) []string {
	d := xml.NewDecoder(bytes.NewReader(b))
	key := ""
	inside := false
	out := []string{}
	for {
		tok, e := d.Token()
		if e == io.EOF {
			break
		}
		must(e)
		switch v := tok.(type) {
		case xml.StartElement:
			switch v.Name.Local {
			case "key":
				must(d.DecodeElement(&key, &v))
			case "array":
				inside = key == "ProgramArguments"
			case "string":
				var s string
				must(d.DecodeElement(&s, &v))
				if inside {
					out = append(out, s)
				}
			}
		case xml.EndElement:
			if v.Name.Local == "array" {
				inside = false
			}
		}
	}
	return out
}
func assertJob(ctx context.Context, a mapping, want []string) string {
	live, e := a.print(ctx)
	must(e)
	if strings.Join(plistArgs(read(a.mapped())), "\x00") != strings.Join(want, "\x00") {
		panic("actual_os_document_identity")
	}
	for _, arg := range want {
		if !bytes.Contains(live, []byte(arg)) {
			panic("actual_launchd_argument_identity")
		}
	}
	intervalPattern := regexp.MustCompile(`<key>\s*StartInterval\s*</key>\s*<integer>([0-9]+)</integer>`)
	interval := intervalPattern.FindSubmatch(read(a.mapped()))
	if len(interval) != 2 {
		panic("actual_owned_plist_interval")
	}
	if !bytes.Contains(live, []byte(a.Label)) || !bytes.Contains(live, []byte("run interval = "+string(interval[1])+" seconds")) || !bytes.Contains(live, []byte("runs = 0")) {
		panic("actual_launchd_timer")
	}
	return hash([]byte(strings.Join(want, "\x00")))
}

type runner struct{ env []string }

func (r runner) Run(ctx context.Context, exe string, args []string) ([]byte, error) {
	return command(ctx, exe, args, r.env)
}
